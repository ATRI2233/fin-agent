3. STRUCTURE OF base.py
--------------------------------------------------------------------------------
  Module docstring: explains the strategy pattern for the four node types
                    (input / output / debate / regular agent) handled by
                    WorkflowEngine.execute_node, and explicitly notes that
                    transaction handling is intentionally NOT centralized.

  Imports:
    from __future__ import annotations
    from abc import ABC, abstractmethod
    from dataclasses import dataclass
    from typing import TYPE_CHECKING, Any
    from sqlalchemy.orm import Session
    from main.framework.core.agent_dispatcher import AgentDispatcher
    if TYPE_CHECKING:
        from main.framework.core.workflow_engine import WorkflowEngine  # circular-safe

  @dataclass
  class NodeContext:
      node: dict
      execution_id: str
      predecessor_ids: list[str]
      params: dict
      results: dict[str, Any]

  @dataclass
  class NodeResult:
      result: dict[str, Any]
      output: dict[str, Any] | None = None
      session_id: str | None = None
      error: str | None = None

  class NodeExecutor(ABC):
      def __init__(self, dispatcher: AgentDispatcher | None = None) -> None:
          self.dispatcher = dispatcher

      @abstractmethod
      async def execute(self, ctx: NodeContext) -> NodeResult:
          raise NotImplementedError

      def _commit(self, db: Session) -> None:
          db.commit()
